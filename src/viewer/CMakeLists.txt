project("viewer")

skt_snake_to_camel("${CMAKE_PROJECT_NAME}" CAMEL_CASE_CMAKE_PROJECT_NAME)

list(APPEND QML_IMPORT_PATH "${CMAKE_CURRENT_BINARY_DIR}/${CAMEL_CASE_CMAKE_PROJECT_NAME}")

qt6_add_qml_module(
    "${PROJECT_NAME}_item"
    "${QML_MODULE_TYPE}"
    URI
        "${CAMEL_CASE_CMAKE_PROJECT_NAME}"
    VERSION
        1.0
    RESOURCE_PREFIX
        "/${CMAKE_PROJECT_NAME}/imports"
    OUTPUT_DIRECTORY
        "${CAMEL_CASE_CMAKE_PROJECT_NAME}"
    IMPORT_PATH
        "${CAMEL_CASE_CMAKE_PROJECT_NAME}"
    DEPENDENCIES
        QtCore
        QtQuick
        QtQuick.Controls
        QtQuick.Window
        QtQuick.Layouts
        QtQuick.Dialogs
        QtQuick3D
)
set_target_properties(
    "${PROJECT_NAME}_item"
    PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${CAMEL_CASE_CMAKE_PROJECT_NAME}"
)
skt_generate_export_header("${PROJECT_NAME}_item" "${PROJECT_NAME}")
target_sources(
    "${PROJECT_NAME}_item"
    PRIVATE
        "${PROJECT_NAME}_export.h"
        "${PROJECT_NAME}.hpp"
        "${PROJECT_NAME}.cpp"
        "task_queue.hpp"
        "task_queue.cpp"
        "render_node.hpp"
        "render_node.cpp"
        "file_io.hpp"
        "file_io.cpp"
        "scenes.hpp"
        "scenes.cpp"
        "engine.hpp"
        "engine.cpp"
        "tree.hpp"
        "tree.cpp"
        "descriptors.hpp"
        "descriptors.cpp"
        "pipelines.hpp"
        "pipelines.cpp"
        "engine_wrapper.hpp"
        "engine_wrapper.cpp"
        "renderer.hpp"
        "renderer.cpp"
        "utils.hpp"
        "utils.cpp"
)
target_link_libraries(
    "${PROJECT_NAME}_item"
    PRIVATE
        Qt6::Core
        Qt6::Gui
        Qt6::GuiPrivate
        Qt6::Qml
        Qt6::Quick
        Qt6::Widgets
        Qt6::Svg
        Qt6::Xml
        Qt6::Quick3D
        Qt6::Concurrent
        "libformat"
        "libengine"
        "libcodegen_vulkan_utils"
        "libscene_data"
        "libscene_loader"
        "libdebug_utils"
        "libcompute"
        "libbuilder"
        "gli"
)
skt_target_shaders(
    "${PROJECT_NAME}_item"
    SHADERS
        "shaders/uniform_buffer.glsl"
        "shaders/barycentric_color.frag.glsl"
        "shaders/fullscreen_rect.vert.glsl"
        "shaders/identity.vert.glsl"
        "shaders/offscreen.frag.glsl"
        "shaders/trace.comp.glsl"
        "shaders/utils.glsl"
    OUTPUT_VARIABLE
        compiled_shaders
)
qt6_target_qml_sources(
    "${PROJECT_NAME}_item"
    RESOURCES
        ${compiled_shaders}
)

qt6_add_executable("${PROJECT_NAME}")
skt_add_executable(
    EXTERNAL
    MAIN_LINK "${PROJECT_NAME}_item"
    SOURCES
        "application.hpp"
        "application.cpp"
    LINKS
        "libcommon"
        "libformat"
        "libengine"
        Qt6::Gui
        Qt6::GuiPrivate
        Qt6::Qml
        Qt6::Widgets
        Qt6::Quick
        Qt6::QuickControls2
)
qt6_add_resources(
    "${PROJECT_NAME}"
    "ui"
    PREFIX
        "/${CMAKE_PROJECT_NAME}"
    FILES
        "qml/components/CenteredDialog.qml"
        "qml/components/CenteredText.qml"
        "qml/components/NumberSpinBox.qml"
        "qml/components/FastFileOpenDialog.qml"
        "qml/components/TaskManager.qml"
        "qml/components/WheelSpinBox.qml"
        "qml/qmldir"
        "qml/ui.qml"
        "qml/utils.js"
)

add_subdirectory("tests")
